+import { z } from 'zod';
+export function trustedPushEndpoint(value: string): boolean {
+  try {
+    const u=new URL(value),h=u.hostname;
+    return u.protocol==='https:' && !u.port && !u.username && !u.password && !u.hash && (
+      h==='fcm.googleapis.com' || h==='updates.push.services.mozilla.com' || h==='push.services.mozilla.com' || h.endsWith('.push.services.mozilla.com') || h.endsWith('.push.apple.com') || h.endsWith('.notify.windows.com')
+    );
+  } catch { return false; }
+}
